四、核心结论

A

券提升提单率 +3.74pp（结论稳健）。

实验组 50.78% vs 对照组 47.04%，第一周 +2.87pp。券的拉量作用在 8 月批次同样存在，但幅度小于 7 月。

B

到期口径下实验组盈利率偏低、逾期率偏高（早期快照，需复核）。

实验组盈利率 5.76% vs 对照 24.97%，逾期率 36.07% vs 25.19%。但仅 30% 订单到期、样本 122/135 件，结论不稳健，观察期成熟后可能收敛。

C

警惕早期快照的「逾期暴露偏差」。

逾期订单（status=8）在数据中即时可见，而正常订单（status=7）需等还款完成才计入「到期」。观察期越短，逾期订单占比越容易被高估，导致盈利率被低估。

附录：完整 PGSQL

以下代码覆盖本报告的汇总、提单率、四周窗口交叉（到期订单 + 分期级口径）。

模块一：构建 8 月召回实验用户（4 批 T4 合并）
-- 数据源：no_apply_0730_0804_30pct / 0731_0805 / 0801_0806 / 0802_0807
-- 召回日 = 表名最后一个日期（如 0730_0804 -> 2026-08-04）
WITH t7 AS (
    SELECT user_id, ab_group, MIN(recall_date) AS recall_date
    FROM (
        SELECT churn_user_id::bigint AS user_id, ab_group, DATE '2026-08-04' AS recall_date
        FROM wangchuanliang.no_apply_0730_0804_30pct
        UNION ALL
        SELECT churn_user_id::bigint, ab_group, DATE '2026-08-05'
        FROM wangchuanliang.no_apply_0731_0805_30pct
        UNION ALL
        SELECT churn_user_id::bigint, ab_group, DATE '2026-08-06'
        FROM wangchuanliang.no_apply_0801_0806_30pct
        UNION ALL
        SELECT churn_user_id::bigint, ab_group, DATE '2026-08-07'
        FROM wangchuanliang.no_apply_0802_0807_30pct
    ) x
    GROUP BY user_id, ab_group
)
SELECT ab_group, COUNT(DISTINCT user_id) AS 用户数
FROM t7
GROUP BY ab_group
ORDER BY ab_group;
模块二：汇总统计（到期订单 + 分期级）
-- 口径：只统计已到期订单（due_date < current_date）；分期级（repayment_plan_v2_copy 每期一行，仅计 status_period in (7,8)）
-- 盈利率 = (Σrepaid_amt_period - Σremit_amt_period) / Σremit_amt_period
WITH t7 AS (
    SELECT user_id, ab_group, MIN(recall_date) AS recall_date
    FROM (
        SELECT churn_user_id::bigint AS user_id, ab_group, DATE '2026-08-04' AS recall_date
        FROM wangchuanliang.no_apply_0730_0804_30pct
        UNION ALL
        SELECT churn_user_id::bigint, ab_group, DATE '2026-08-05' FROM wangchuanliang.no_apply_0731_0805_30pct
        UNION ALL
        SELECT churn_user_id::bigint, ab_group, DATE '2026-08-06' FROM wangchuanliang.no_apply_0801_0806_30pct
        UNION ALL
        SELECT churn_user_id::bigint, ab_group, DATE '2026-08-07' FROM wangchuanliang.no_apply_0802_0807_30pct
    ) x
    GROUP BY user_id, ab_group
)
SELECT
    t7.ab_group                                                             AS 分组,
    COUNT(DISTINCT a.serial_id)                                             AS 到期件数,
    COUNT(DISTINCT a.user_id)                                               AS 到期人数,
    ROUND(ROUND(SUM(a.repaid_amt_period) - SUM(a.remit_amt_period), 2)
          / NULLIF(ROUND(SUM(a.remit_amt_period), 2), 0) * 100, 2)          AS 盈利率,
    ROUND((SUM(CASE WHEN a.loan_status_code_period = 8 THEN a.remit_amt_period ELSE 0 END) + 0.0)
          / NULLIF(SUM(a.remit_amt_period), 0) * 100, 2)                    AS 逾期率_金额,
    ROUND((COUNT(CASE WHEN a.loan_status_code_period = 8 THEN a.serial_id END) + 0.0)
          / NULLIF(COUNT(a.serial_id), 0) * 100, 2)                         AS 逾期率_分期,
    ROUND((COUNT(DISTINCT CASE WHEN a.loan_status_code_period = 8 THEN a.serial_id END) + 0.0)
          / NULLIF(COUNT(DISTINCT a.serial_id), 0) * 100, 2)                AS 逾期率_订单
FROM t7
JOIN wangchuanliang.repayment_plan_v2_copy a
  ON a.user_id = t7.user_id AND a.remit_date >= t7.recall_date
WHERE a.is_remit = 1 AND a.due_date < current_date AND a.loan_status_code_period IN (7, 8)
GROUP BY t7.ab_group
ORDER BY t7.ab_group;
模块三：提单率（召回后累计）
WITH t7 AS (
    SELECT user_id, ab_group, MIN(recall_date) AS recall_date
    FROM (
        SELECT churn_user_id::bigint AS user_id, ab_group, DATE '2026-08-04' AS recall_date
        FROM wangchuanliang.no_apply_0730_0804_30pct
        UNION ALL
        SELECT churn_user_id::bigint, ab_group, DATE '2026-08-05' FROM wangchuanliang.no_apply_0731_0805_30pct
        UNION ALL
        SELECT churn_user_id::bigint, ab_group, DATE '2026-08-06' FROM wangchuanliang.no_apply_0801_0806_30pct
        UNION ALL
        SELECT churn_user_id::bigint, ab_group, DATE '2026-08-07' FROM wangchuanliang.no_apply_0802_0807_30pct
    ) x
    GROUP BY user_id, ab_group
)
SELECT
    t7.ab_group                                                             AS 分组,
    COUNT(DISTINCT t7.user_id)                                              AS 用户数,
    COUNT(DISTINCT CASE WHEN o.serial_id IS NOT NULL THEN t7.user_id END)   AS 提单用户,
    ROUND(COUNT(DISTINCT CASE WHEN o.serial_id IS NOT NULL THEN t7.user_id END) * 100.0
          / NULLIF(COUNT(DISTINCT t7.user_id), 0), 2)                       AS 提单率
FROM t7
LEFT JOIN wangchuanliang.order_loan_f_v2_copy o
       ON o.user_id = t7.user_id AND o.apply_date >= t7.recall_date
GROUP BY t7.ab_group
ORDER BY t7.ab_group;
模块四：AB 组 × 四周窗口交叉（到期口径）
-- 第一周=召回日~+6；第二周=+7~+13；第三周=+14~+20；21天后=+21 起
WITH t7 AS (
    SELECT user_id, ab_group, MIN(recall_date) AS recall_date
    FROM (
        SELECT churn_user_id::bigint AS user_id, ab_group, DATE '2026-08-04' AS recall_date
        FROM wangchuanliang.no_apply_0730_0804_30pct
        UNION ALL
        SELECT churn_user_id::bigint, ab_group, DATE '2026-08-05' FROM wangchuanliang.no_apply_0731_0805_30pct
        UNION ALL
        SELECT churn_user_id::bigint, ab_group, DATE '2026-08-06' FROM wangchuanliang.no_apply_0801_0806_30pct
        UNION ALL
        SELECT churn_user_id::bigint, ab_group, DATE '2026-08-07' FROM wangchuanliang.no_apply_0802_0807_30pct
    ) x
    GROUP BY user_id, ab_group
),
ords AS (
    SELECT t7.ab_group,
           CASE WHEN a.apply_date BETWEEN t7.recall_date AND t7.recall_date + 6 THEN '第一周'
                WHEN a.apply_date BETWEEN t7.recall_date + 7 AND t7.recall_date + 13 THEN '第二周'
                WHEN a.apply_date BETWEEN t7.recall_date + 14 AND t7.recall_date + 20 THEN '第三周'
                ELSE '21天后' END                                             AS win,
           a.serial_id, a.user_id, a.loan_status_code_period, a.remit_amt_period, a.repaid_amt_period
    FROM t7
    JOIN wangchuanliang.repayment_plan_v2_copy a
      ON a.user_id = t7.user_id AND a.remit_date >= t7.recall_date
    WHERE a.is_remit = 1 AND a.due_date < current_date AND a.loan_status_code_period IN (7, 8)
)
SELECT
    ab_group                                                                AS 分组,
    win                                                                     AS 窗口,
    COUNT(DISTINCT serial_id)                                               AS 到期件数,
    ROUND(ROUND(SUM(repaid_amt_period) - SUM(remit_amt_period), 2)
          / NULLIF(ROUND(SUM(remit_amt_period), 2), 0) * 100, 2)            AS 盈利率,
    ROUND((SUM(CASE WHEN loan_status_code_period = 8 THEN remit_amt_period ELSE 0 END) + 0.0)
          / NULLIF(SUM(remit_amt_period), 0) * 100, 2)                      AS 逾期率_金额,
    ROUND((COUNT(DISTINCT CASE WHEN loan_status_code_period = 8 THEN serial_id END) + 0.0)
          / NULLIF(COUNT(DISTINCT serial_id), 0) * 100, 2)                  AS 逾期率_订单
FROM ords
GROUP BY ab_group, win
ORDER BY ab_group,
         CASE win WHEN '第一周' THEN 1 WHEN '第二周' THEN 2 WHEN '第三周' THEN 3 ELSE 4 END;
模块五：各窗口提单用户数与历史借款次数
-- 各窗口提单用户数（count distinct user）、历史借款次数（total_remit_rnb 均值）
WITH feat AS (
    SELECT churn_user_id::bigint AS user_id, ab_group, DATE '2026-08-04' AS recall_date, total_remit_rnb
    FROM wangchuanliang.no_apply_0730_0804_30pct
    UNION ALL
    SELECT churn_user_id::bigint, ab_group, DATE '2026-08-05', total_remit_rnb FROM wangchuanliang.no_apply_0731_0805_30pct
    UNION ALL
    SELECT churn_user_id::bigint, ab_group, DATE '2026-08-06', total_remit_rnb FROM wangchuanliang.no_apply_0801_0806_30pct
    UNION ALL
    SELECT churn_user_id::bigint, ab_group, DATE '2026-08-07', total_remit_rnb FROM wangchuanliang.no_apply_0802_0807_30pct
),
win_users AS (
    SELECT ab_group,
           CASE WHEN o.apply_date BETWEEN f.recall_date AND f.recall_date + 6 THEN 'W1'
                WHEN o.apply_date BETWEEN f.recall_date + 7 AND f.recall_date + 13 THEN 'W2'
                WHEN o.apply_date BETWEEN f.recall_date + 14 AND f.recall_date + 20 THEN 'W3'
                ELSE 'W4' END                                                   AS win,
           f.user_id, f.total_remit_rnb
    FROM feat f
    JOIN wangchuanliang.order_loan_f_v2_copy o
      ON o.user_id = f.user_id AND o.apply_date >= f.recall_date
)
SELECT
    ab_group                                                                    AS 分组,
    win                                                                         AS 窗口,
    COUNT(DISTINCT user_id)                                                     AS 提单用户,
    ROUND(AVG(total_remit_rnb)::numeric, 2)                                     AS 历史借款次数
FROM win_users
GROUP BY ab_group, win
ORDER BY ab_group, CASE win WHEN 'W1' THEN 1 WHEN 'W2' THEN 2 WHEN 'W3' THEN 3 ELSE 4 END;
模块六：基线特征（历史借款次数 / 高息率）
-- 实验开始前的用户画像基线，用于验证 AB 分组平衡性
SELECT
    ab_group                                                                    AS 分组,
    ROUND(AVG(total_remit_rnb)::numeric, 2)                                     AS 基线借款次数,
    ROUND(AVG(high_interest_rate)::numeric, 3)                                  AS 基线高息率
FROM (
    SELECT churn_user_id::bigint AS user_id, ab_group, total_remit_rnb, high_interest_rate
    FROM wangchuanliang.no_apply_0730_0804_30pct
    UNION ALL
    SELECT churn_user_id::bigint, ab_group, total_remit_rnb, high_interest_rate FROM wangchuanliang.no_apply_0731_0805_30pct
    UNION ALL
    SELECT churn_user_id::bigint, ab_group, total_remit_rnb, high_interest_rate FROM wangchuanliang.no_apply_0801_0806_30pct
    UNION ALL
    SELECT churn_user_id::bigint, ab_group, total_remit_rnb, high_interest_rate FROM wangchuanliang.no_apply_0802_0807_30pct
) x
GROUP BY ab_group
ORDER BY ab_group;
口径说明：4 批 T4 用户合并（召回日 08-04 / 08-05 / 08-06 / 08-07）。只统计已到期订单（due_date < current_date），分期级统计（repayment_plan_v2_copy 每期一行，仅计 loan_status_code_period ∈ (7,8)）。盈利率 = (Σrepaid_amt_period − Σremit_amt_period) / Σremit_amt_period；逾期率_金额/分期/订单 三个口径。
